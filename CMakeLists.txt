cmake_minimum_required(VERSION 3.20)
project(kairocopy CXX C RC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# a release: no MinGW runtime DLLs (CMake ends its static list with -Bdynamic; back to static first)
option(KAIROCOPY_STATIC "link statically (a release)" OFF)
if(KAIROCOPY_STATIC)
  set(CMAKE_CXX_STANDARD_LIBRARIES "-Wl,-Bstatic ${CMAKE_CXX_STANDARD_LIBRARIES}")
endif()

add_library(kccore STATIC src/win/sys.cpp src/core/rules.cpp src/core/manifest.cpp src/core/plan.cpp src/core/copier.cpp)
target_include_directories(kccore PUBLIC src)
target_compile_definitions(kccore PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(kccore PRIVATE -Wall -Wextra -Wno-unused-parameter)
if(KAIROCOPY_STATIC)
  target_link_libraries(kccore PUBLIC libsqlite3.a libxxhash.a)
  target_link_options(kccore PUBLIC -static)
else()
  target_link_libraries(kccore PUBLIC sqlite3 xxhash)
endif()

add_executable(kairocopy-cli src/cli/main.cpp src/win/manifest.rc)
target_link_libraries(kairocopy-cli PRIVATE kccore)
target_compile_options(kairocopy-cli PRIVATE -Wall -Wextra -Wno-unused-parameter)

add_executable(kc_tests tests/tests.cpp)
target_link_libraries(kc_tests PRIVATE kccore)
